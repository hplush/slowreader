CI verified build reproducibility

# Release with this commit
TAG=TAG
FILES=https://github.com/hplush/slowreader/releases/download/$TAG

# Image digest and maintainer’s hardware key signature of it
curl -fsSLO $FILES/image.txt
curl -fsSLO $FILES/image.txt.sig
curl -fsSLO https://gitlab.com/hplush/slowreader-verifier/-/raw/main/allowed_signers
ssh-keygen -Y verify -f allowed_signers -I release -n slowreader-release \
  -s image.txt.sig < image.txt

# GitHub signature, that our release workflow built it from the tag
gh attestation verify oci://$(cat image.txt) --repo hplush/slowreader \
  --source-ref refs/tags/$TAG

# Rebuild the image from the same commit and base images
docker buildx create --use --driver docker-container
docker buildx replay build $(cat image.txt) \
  --output type=oci,dest=replay.tar,rewrite-timestamp=true
mkdir replay && tar -xf replay.tar -C replay
REBUILT=$(jq -r '.manifests[0].digest' replay/index.json)
jq -r .config.digest replay/blobs/sha256/${REBUILT#sha256:}

# Config digest of the published image must be the same
PLATFORM=$(docker buildx imagetools inspect --raw $(cat image.txt) \
  | jq -r '.manifests[] | select(.platform.architecture == "amd64") | .digest')
docker buildx imagetools inspect --raw ghcr.io/hplush/slowreader@$PLATFORM \
  | jq -r .config.digest

Assets match hashes

# Hash every file from manifest.json and compare
curl -fsS https://slowreader.app/manifest.json -o site-manifest.json
for url in $(jq -r '.integrity | keys[]' site-manifest.json); do
  hash="sha256-$(curl -fsS https://slowreader.app$url \
    | openssl dgst -sha256 -binary | base64)"
  expected=$(jq -r --arg url "$url" '.integrity[$url]' site-manifest.json)
  [ "$hash" = "$expected" ] && echo "✓ $url" || echo "✗ $url"
done

# Users open pages by short address, check them too
for page in / /welcome; do
  curl -fsS https://slowreader.app$page | openssl dgst -sha256 -binary | base64
done
